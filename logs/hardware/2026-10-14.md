## 06:30: Listen on the serial port for 16 seconds

Command:
```sh
python3 - <<'EOF'
import os, termios, time, select
fd = os.open('/dev/cu.usbmodem1403', os.O_RDONLY | os.O_NOCTTY | os.O_NONBLOCK)
a = termios.tcgetattr(fd)
a[0]=0; a[1]=0; a[2]=termios.CS8|termios.CREAD|termios.CLOCAL; a[3]=0
a[4]=a[5]=termios.B115200
termios.tcsetattr(fd, termios.TCSANOW, a)
end=time.time()+16; buf=b''
while time.time()<end:
    r,_,_=select.select([fd],[],[],0.5)
    if r:
        try: buf+=os.read(fd,4096)
        except BlockingIOError: pass
print(buf.decode(errors='replace') or '(nothing received)')
EOF
```

## 06:39: Read 17 seconds of serial output after reset

Command:
```sh
sleep 2; python3 - <<'EOF'
import os, termios, time, select, glob
port = None
for _ in range(20):
    g = glob.glob('/dev/cu.usbmodem*')
    if g: port = g[0]; break
    time.sleep(0.5)
fd = os.open(port, os.O_RDONLY | os.O_NOCTTY | os.O_NONBLOCK)
a = termios.tcgetattr(fd)
a[0]=0; a[1]=0; a[2]=termios.CS8|termios.CREAD|termios.CLOCAL; a[3]=0
a[4]=a[5]=termios.B115200
termios.tcsetattr(fd, termios.TCSANOW, a)
end=time.time()+17; buf=b''
while time.time()<end:
    r,_,_=select.select([fd],[],[],0.5)
    if r:
        try: buf+=os.read(fd,4096)
        except (BlockingIOError, OSError): pass
print(port); print(buf.decode(errors='replace') or '(nothing received)')
EOF
```
