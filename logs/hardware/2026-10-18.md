## 14:03: Run the MCP2515 test at 2 MHz and 31 kHz

Command:
```sh
SP=<scratch>; cat > $SP/diag/candiag2.c <<'EOF'
/* Temporary MCP2515 diagnostic at two SPI speeds (not part of the repo). */
#include "board.h"

static uint8_t rd(uint8_t a)
{
    uint8_t cmd[2] = {0x03, a}, rx = 0;
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);
    HAL_SPI_Transmit(&board_spi1, cmd, 2, 10);
    HAL_SPI_Receive(&board_spi1, &rx, 1, 10);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_SET);
    return rx;
}

static void wr(uint8_t a, uint8_t v)
{
    uint8_t cmd[3] = {0x02, a, v};
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);
    HAL_SPI_Transmit(&board_spi1, cmd, 3, 10);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_SET);
}

static void speed(unsigned br) /* SPI clock = 4 MHz / 2^(br+1) */
{
    SPI1->CR1 &= ~SPI_CR1_SPE;
    SPI1->CR1 = (SPI1->CR1 & ~(7u << 3)) | (br << 3);
    SPI1->CR1 |= SPI_CR1_SPE;
}

static void hex(uint8_t v)
{
    uart_puts(" ");
    uart_put_hex(v, 2);
}

int main(void)
{
    board_init();
    HAL_Delay(300);
    for (int round = 0; round < 4; round++) {
        for (unsigned br = 0; br <= 6; br += 6) { /* 2 MHz, then 31 kHz */
            speed(br);
            uint8_t rst = 0xC0;
            HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_RESET);
            HAL_SPI_Transmit(&board_spi1, &rst, 1, 10);
            HAL_GPIO_WritePin(GPIOB, GPIO_PIN_5, GPIO_PIN_SET);
            HAL_Delay(5);
            GPIOA->PUPDR = (GPIOA->PUPDR & ~(3u << 12)) | (1u << 12); /* MISO pull-up */
            uart_puts(br ? "DIAG2  31 kHz:" : "DIAG2   2 MHz:");
            uart_puts(" CANSTAT x4");
            for (int i = 0; i < 4; i++) hex(rd(0x0E));
            uart_puts(" | CANCTRL");
            hex(rd(0x0F));
            wr(0x2A, 0x55);
            uart_puts(" | CNF1 w55");
            hex(rd(0x2A));
            wr(0x2A, 0xAA);
            uart_puts(" wAA");
            hex(rd(0x2A));
            wr(0x2A, 0x00);
            uart_puts("   (want 80 80 80 80 | 87 | 55 AA)\n");
        }
        HAL_Delay(1000);
    }
    for (;;) {
    }
}
EOF
cd <repo>/boards/nucleo_wl55jc && R=../.. && arm-none-eabi-gcc -mcpu=cortex-m4 -mthumb -mfloat-abi=soft -Os -std=c99 -Wall -ffreestanding -ffunction-sections -fdata-sections -fno-tree-loop-distribute-patterns -DCORE_CM4 -DSTM32WL55xx -DFHSS_HAL_HEADER='"hal_shim.h"' -I. -Ilibc -I$R/include -I$R/third_party/cmsis_core -I$R/third_party/cmsis_device_wl startup.c board.c hal_shim.c libc_min.c $SP/diag/candiag2.c -T stm32wl55jc_cm4.ld -nostdlib -Wl,--gc-sections -lgcc -o $SP/diag/candiag2.elf 2>&1 | head -5; arm-none-eabi-objcopy -O binary $SP/diag/candiag2.elf $SP/diag/candiag2.bin && (python3 $SP/cap.py 9 > $SP/diag5.txt &) ; sleep 1; cat $SP/diag/candiag2.bin > /Volumes/NOD_WL55JC/candiag2.bin; sync; sleep 9; grep -a DIAG2 $SP/diag5.txt | head -8
```
