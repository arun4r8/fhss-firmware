## 11:03: Analyze delivery between the two boards

Command:
```sh
SP=<scratch>; python3 - <<EOF
import re
g=[l.split(',') for l in open('$SP/ground.txt',errors='replace').read().splitlines() if l.startswith(('RX,','HOP,'))]
p=[l.split(',') for l in open('$SP/pod.txt',errors='replace').read().splitlines() if l.startswith('TX,')]
rx=[(int(x[1]),int(x[2]),int(x[3])) for x in g if x[0]=='RX' and len(x)==4]
hops=[(int(x[1]),int(x[2]),int(x[3]),x[4]) for x in g if x[0]=='HOP' and len(x)==5]
tx=[(int(x[1]),int(x[2]),int(x[3])) for x in p if len(x)==5]
print('ground RX packets:',len(rx),' first at ground ms',rx[0][0],' last',rx[-1][0])
# gaps in consecutive hop indices while locked
missing=0
for a,b in zip(rx,rx[1:]):
    d=(b[1]-a[1])%50
    if d!=1: missing+=d-1
print('hops with no packet between first and last RX:',missing)
locked=[h for h in hops if h[3]=='H']
from itertools import groupby
iv=[b[0]-a[0] for a,b in zip(locked,locked[1:])]
print('ground hops while locked:',len(locked),' dwell min/max:',min(iv),max(iv))
# channels: ground RX channel equals the channel the pod used for that hop index
podch={}
for t,h,c in tx: podch[h]=c
mism=sum(1 for t,h,c in rx if h in podch and podch[h]!=c)
print('RX on a different channel than pod used for that hop:',mism)
# rx interval regularity
ri=[b[0]-a[0] for a,b in zip(rx,rx[1:])]
print('RX spacing ms min/max:',min(ri),max(ri))
EOF
grep "^status" $SP/ground.txt | tail -1
```
